const SHEET_SESSIONS = 'Sessions'; const SHEET_RESPONSES = 'Responses'; const SHEET_REPORTS = 'Reports_Summary'; function getOrCreateSheet(ss, name, headers) { let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); } if (headers && headers.length > 0) { const firstRow = sheet.getRange(1, 1, 1, headers.length).getValues()[0]; const isEmpty = firstRow.every(v => v === '' || v === null); if (isEmpty) { sheet.getRange(1, 1, 1, headers.length).setValues([headers]); sheet.setFrozenRows(1); } } return sheet; } function doGet(e) { try { const params = (e && e.parameter) || {}; const action = params.action || ''; const sessionId = String(params.id || '').trim().toLowerCase(); const ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'getLatestSession') { const sheet = ss.getSheetByName(SHEET_SESSIONS); if (!sheet || sheet.getLastRow() < 2) { return jsonOut({ ok: false, session: null }); } const rows = sheet.getDataRange().getValues(); for (let i = rows.length - 1; i >= 1; i--) { let sessionObj = null; try { sessionObj = JSON.parse(rows[i][7]); } catch (err) {} if (sessionObj && sessionObj.id) { return jsonOut({ ok: true, session: sessionObj }); } } return jsonOut({ ok: false, session: null }); } if (action === 'getSession') { const sheet = ss.getSheetByName(SHEET_SESSIONS); if (!sheet || sheet.getLastRow() < 2) { return jsonOut({ ok: false, session: null }); } const rows = sheet.getDataRange().getValues(); for (let i = rows.length - 1; i >= 1; i--) { if (String(rows[i][0] || '').trim().toLowerCase() === sessionId) { let sessionObj = null; try { sessionObj = JSON.parse(rows[i][7]); } catch (err) {} return jsonOut({ ok: true, session: sessionObj }); } } return jsonOut({ ok: false, session: null }); } if (action === 'getResults') { const sheet = ss.getSheetByName(SHEET_RESPONSES); if (!sheet || sheet.getLastRow() < 2) { return jsonOut({ ok: true, responses: [] }); } const rows = sheet.getDataRange().getValues(); const responses = []; for (let i = 1; i < rows.length; i++) { const rowSid = String(rows[i][0] || '').trim().toLowerCase(); if (!sessionId || rowSid === sessionId) { let answersArr = []; try { answersArr = JSON.parse(rows[i][10] || '[]'); } catch (err) {} responses.push({ id: String(rows[i][1] || `RESP-${i}`), participantLabel: String(rows[i][5] || 'Учасник'), rank: String(rows[i][6] || 'Приховано'), correctCount: Number(rows[i][7] || 0), totalCount: Number(rows[i][8] || 0), scorePct: Number(rows[i][9] || 0), answers: answersArr, riskLevel: String(rows[i][11] || 'med'), submittedAt: String(rows[i][12] || ''), durationSec: Number(rows[i][13] || 0) }); } } return jsonOut({ ok: true, responses: responses }); } return jsonOut({ ok: true, message: 'Portal API active' }); } catch (err) { return jsonOut({ ok: false, error: String(err) }); } } function doPost(e) { try { const raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}'; const data = JSON.parse(raw); const action = data.action || ''; const ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'createSession') { const s = data.session || {}; const sheet = getOrCreateSheet(ss, SHEET_SESSIONS, [ 'Session_ID', 'Created_At', 'Military_Unit', 'Unit', 'Topic', 'Questions_Count', 'Anonymity', 'Session_JSON' ]); sheet.appendRow([ s.id || '', s.createdAt || new Date().toISOString(), s.militaryUnit || '', s.unit || '', s.topicTitle || '', s.qCount || 0, s.anonymity || 'anonymous', JSON.stringify(s) ]); return jsonOut({ ok: true, action: 'createSession' }); } if (action === 'submitResponse') { const r = data.response || {}; const sheet = getOrCreateSheet(ss, SHEET_RESPONSES, [ 'Session_ID', 'Response_ID', 'Server_Timestamp', 'Military_Unit', 'Unit', 'Participant', 'Rank', 'Correct_Count', 'Total_Count', 'Score_Pct', 'Answers_JSON', 'Risk_Level', 'Submitted_Time', 'Duration_Sec' ]); sheet.appendRow([ data.sessionId || '', r.id || '', new Date().toISOString(), data.militaryUnit || '', data.unit || '', r.participantLabel || '', r.rank || '', r.correctCount || 0, r.totalCount || 0, r.scorePct || 0, JSON.stringify(r.answers || []), r.riskLevel || 'med', r.submittedAt || '', r.durationSec || 0 ]); return jsonOut({ ok: true, action: 'submitResponse' }); } if (action === 'syncReport') { const sheet = getOrCreateSheet(ss, SHEET_REPORTS, [ 'Updated_At', 'Session_ID', 'Military_Unit', 'Unit', 'Topic', 'Submitted_Count', 'Avg_Score_Pct', 'High_Risk_Count', 'Problematic_Questions_And_Solutions', 'Recommendations' ]); sheet.appendRow([ new Date().toISOString(), data.sessionId || '', data.militaryUnit || '', data.unit || '', data.topicTitle || '', data.submittedCount || 0, data.avgScore || 0, data.highRiskCount || 0, data.problemsText || '', data.recommendationsText || '' ]); return jsonOut({ ok: true, action: 'syncReport' }); } return jsonOut({ ok: false, error: 'Unknown action' }); } catch (err) { return jsonOut({ ok: false, error: String(err) }); } } function jsonOut(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
